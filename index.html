/** ======================= * KONSTANTA * ======================= */ const SPREADSHEET_ID = "1mxOEhWCiHv1OoyIUlJq2Xq13smO1iIEg09oGxljHUt4"; const DATA_DROPDOWN_SHEET = "DataDropdown"; const TUGAS_SHEET = "Tugas"; const ROOT_FOLDER_ID = "1wfSlXjZ4E1cuk6DGYwo16LJLACU52qAn"; const MAX_FILE_SIZE = 20 * 1024 * 1024; // 20MB const ALLOWED_TYPES = [ "application/pdf", "image/png", "image/jpeg", "application/msword", "application/vnd.openxmlformats-officedocument.wordprocessingml.document", "application/vnd.ms-excel", "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", "application/vnd.ms-powerpoint", "application/vnd.openxmlformats-officedocument.presentationml.presentation", "application/zip", "application/x-rar-compressed" ]; /** ======================= * Tampilkan halaman HTML * ======================= */ function doGet() { return HtmlService .createTemplateFromFile("Form") .evaluate() .setTitle("Upload Tugas") .addMetaTag("viewport", "width=device-width, initial-scale=1") .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } /** ======================= * Ambil data dropdown dari sheet * ======================= */ function getDropdownData() { const sheet = SpreadsheetApp .openById(SPREADSHEET_ID) .getSheetByName(DATA_DROPDOWN_SHEET); const values = sheet.getDataRange().getValues().slice(1); let namaKelasMap = {}; let mapelSet = new Set(); let tugasSet = new Set(); values.forEach(([kelas, namaKelas, mapel, tugas]) => { if (kelas && namaKelas) { if (!namaKelasMap[kelas]) namaKelasMap[kelas] = []; if (!namaKelasMap[kelas].includes(namaKelas)) namaKelasMap[kelas].push(namaKelas); } if (mapel) mapelSet.add(mapel); if (tugas) tugasSet.add(tugas); }); return { namaKelasMap, mapel: Array.from(mapelSet), tugas: Array.from(tugasSet) }; } /** ======================= * Upload File * ======================= */ function uploadFile(data) { const lock = LockService.getScriptLock(); try { lock.waitLock(30000); if (!data.file || !data.file.data) throw new Error("File kosong atau corrupt."); const fileBytes = Utilities.base64Decode(data.file.data); validateData(data, fileBytes); const sheet = getOrCreateSheet(TUGAS_SHEET, [ "Timestamp", "Nama Siswa", "Kelas", "Nama Kelas", "Mapel", "Jenis Tugas", "Nama File", "File URL", "Jenis Tugas", "File URL", "Email", "File ID" ]); const folder = createNestedFolder(data); const existingRow = findExistingRow(sheet, data); // Hapus file lama jika ada if (existingRow > 0) deleteOldFileById(sheet, existingRow); // Simpan file baru const fileName = createUniqueFileName(data.email, data.file.name); const file = saveFileToFolder(fileBytes, data.file.type, fileName, folder); // Update sheet saveRow(sheet, existingRow, data, file.getUrl(), fileName, file.getId()); return { status: "success", url: file.getUrl(), replaced: existingRow > 0 }; } catch (e) { logError(e.message, data); return { status: "error", message: e.message }; } finally { try { lock.releaseLock(); } catch (err) {} } } /** ======================= * Validasi Data & File * ======================= */ function validateData(data, fileBytes) { if ( !data.nama || !data.email || !data.kelas || !data.namaKelas || !data.mapel || !data.tugas ) throw new Error("Data tidak lengkap."); if (!ALLOWED_TYPES.includes(data.file.type)) throw new Error("Tipe file tidak diizinkan."); if (fileBytes.length > MAX_FILE_SIZE) throw new Error("Ukuran file melebihi 20MB."); } /** ======================= * Buat nama file unik * ======================= */ function createUniqueFileName(email, fileName) { const timestamp = new Date().getTime(); const safeEmail = email.replace(/[^a-zA-Z0-9]/g, "_"); const safeFileName = fileName.replace(/[^a-zA-Z0-9_\-\.]/g, "_"); //return `${safeEmail}_${timestamp}_${safeFileName}`; return `${safeFileName}_${safeEmail}_${timestamp}`; } /** ======================= * Hapus file lama berdasarkan File ID * ======================= */ function deleteOldFileById(sheet, rowIndex) { const oldFileId = sheet.getRange(rowIndex, 10).getValue(); if (!oldFileId) return; try { const file = DriveApp.getFileById(oldFileId); file.setTrashed(true); } catch (e) { /* ignore jika file tidak ada */ } } /** ======================= * Save / Update Row * ======================= */ function saveRow(sheet, rowIndex, data, fileUrl, fileName, fileId) { const row = [ new Date(), data.nama, data.kelas, data.namaKelas, data.mapel, data.tugas, fileName, `=HYPERLINK("${fileUrl}";"Lihat File")`, data.email, fileId ]; if (rowIndex > 0) sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]); else sheet.appendRow(row); } /** ======================= * Folder Nested * ======================= */ function createNestedFolder(data) { const root = DriveApp.getFolderById(ROOT_FOLDER_ID); return [data.kelas, data.namaKelas, data.mapel, data.tugas].reduce( (parent, name) => { const safeName = name.replace(/[^a-zA-Z0-9_\- ]/g, "_"); const folders = parent.getFoldersByName(safeName); return folders.hasNext() ? folders.next() : parent.createFolder(safeName); }, root ); } /** ======================= * Cari Row Existing * ======================= */ function findExistingRow(sheet, data) { const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { const r = rows[i]; if ( r[1] === data.nama && r[2] === data.kelas && r[3] === data.namaKelas && r[4] === data.mapel && r[5] === data.tugas && r[8] === data.email ) return i + 1; } return -1; } /** ======================= * Simpan File * ======================= */ function saveFileToFolder(bytes, type, fileName, folder) { const blob = Utilities.newBlob(bytes, type, fileName); return folder.createFile(blob); } /** ======================= * Sheet Helper * ======================= */ function getOrCreateSheet(name, header) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name).appendRow(header); return sheet; } /** ======================= * Error Logging * ======================= */ function logError(message, data) { try { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName("ErrorLog"); if (!sheet) sheet = ss.insertSheet("ErrorLog").appendRow([ "Timestamp", "Message", "Data" ]); sheet.appendRow([new Date(), message, JSON.stringify(data)]); } catch (e) { console.error("Gagal log error:", message, data); } }
